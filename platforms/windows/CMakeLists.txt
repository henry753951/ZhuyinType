cmake_minimum_required(VERSION 3.24)
project(ZhuyinTypeTsf LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "需要 Windows x64 與 MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /WX /utf-8 /EHsc)
add_library(zt-native STATIC src/BrokerClient.cpp src/Keyboard.cpp src/Composition.cpp src/InputScopeGuid.cpp)
target_include_directories(zt-native PUBLIC src)
target_link_libraries(zt-native PUBLIC ole32 oleaut32 uuid advapi32 user32 windowsapp shcore)
add_library(zhuyintype-tsf SHARED src/TextService.cpp src/InputModeButton.cpp src/Module.cpp src/Registration.cpp src/exports.def src/resources.rc)
target_link_libraries(zhuyintype-tsf PRIVATE zt-native gdi32)
add_executable(zt-tsf-register src/RegisterMain.cpp src/Registration.cpp)
target_link_libraries(zt-tsf-register PRIVATE zt-native)
include(CTest)
if(BUILD_TESTING)
    add_executable(zt-pipe-test tests/PipeTest.cpp)
    target_link_libraries(zt-pipe-test PRIVATE zt-native)
    add_test(NAME pipe-contract COMMAND zt-pipe-test)
    set_tests_properties(pipe-contract PROPERTIES TIMEOUT 30)
    add_executable(zt-tsf-integration tests/TextControl.cpp)
    target_link_libraries(zt-tsf-integration PRIVATE zt-native gdi32 uiautomationcore)
    add_executable(zt-stock-control tests/StockControl.cpp)
    target_link_libraries(zt-stock-control PRIVATE zt-native gdi32 imm32)
    set(ZT_QT_STATE_LIBRARY "" CACHE FILEPATH "供同執行緒測試使用的共用 desktop_state.lib")
    if(ZT_QT_STATE_LIBRARY)
        find_package(Qt6 6.8 REQUIRED COMPONENTS Quick Network)
        add_executable(zt-qt-selfhost tests/QtSelfHost.cpp)
        set_property(TARGET zt-qt-selfhost PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
        target_include_directories(zt-qt-selfhost PRIVATE ../qt/src src ../../engine/crates/zhuyin-ffi/include)
        target_link_libraries(zt-qt-selfhost PRIVATE "${ZT_QT_STATE_LIBRARY}" Qt6::Quick Qt6::Network ole32 oleaut32 uuid advapi32 user32)
    endif()
endif()
install(TARGETS zhuyintype-tsf zt-tsf-register RUNTIME DESTINATION . LIBRARY DESTINATION .)
